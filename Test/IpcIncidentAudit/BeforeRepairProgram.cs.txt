// Characterization, NOT fixed-behavior regression tests: PASS means the observed defect/control
// is reproducible in the linked production sources. UI/game/IPC transport are intentionally inert.
var failures = 0;
var count = 0;
async Task Check(string name, Func<Task> test)
{
    count++;
    PreemptionGate.Disarm();
    CancellationContext.Instance.Set();
    RunnerContext.Instance.taskProgress = null;
    RunnerContext.Instance.IsContinuousRunGroup = false;
    TaskContext.Instance().Config.SuspendedTaskContext = null;
    TaskContext.Instance().CurrentScriptProject = null;
    ExternalInterfaceEventHub.Instance.OnSlotReleased = null;
    App.ServiceProvider = new ServiceCollection().BuildServiceProvider();
    try { Assert(TaskControl.TaskSemaphore.CurrentCount == 1, "slot leaked"); await test(); Console.WriteLine($"PASS {name}"); }
    catch (Exception e) { failures++; Console.WriteLine($"FAIL {name}: {e}"); }
}
static void Assert(bool condition, string message) { if (!condition) throw new Exception(message); }
static Task Done() => Task.CompletedTask;
static string Unique() => Guid.NewGuid().ToString("N");

await Check("D01 stale friendship saved by unnamed UID admission", async () =>
{
    RunnerContext.Instance.taskProgress = new() { CurrentScriptGroupName = "好感任务" };
    PreemptionGate.Arm();
    var ran = 0;
    var result = await new TaskRunner().RunCurrentAsync(() => { ran++; return Done(); });
    var saved = TaskContext.Instance().Config.SuspendedTaskContext;
    Assert(result == TaskRunResult.Preempted && ran == 0, "UID unexpectedly executed");
    Assert(saved?.TaskType == "group" && saved.GroupName == "好感任务" && saved.TaskIndex == 0, "stale restore not reproduced");
});
await Check("D02 Ext parent does not authorize its OneDragonInternal children", async () =>
{
    var parent = JobRegistry.Instance.Submit(JobKind.OneDragon, Unique(), JobSource.Ext).Job;
    JobRegistry.Instance.TryMarkRunning(parent.JobId);
    PreemptionGate.Arm();
    var ran = 0;
    for (var i = 0; i < 2; i++)
    {
        var result = await new TaskRunner().RunCurrentAsync(() => { ran++; return Done(); },
            job: new(JobKind.Group, Unique(), JobSource.OneDragonInternal, ParentJobId: parent.JobId));
        Assert(result == TaskRunResult.Preempted, "child not blocked");
    }
    Assert(ran == 0 && !CancellationContext.Instance.WasCancelled && PreemptionGate.IsArmed,
        "fake-success preconditions differ");
    // Models the handler's existing terminal expression, not execution of the WPF handler itself.
    JobRegistry.Instance.TryMarkTerminal(parent.JobId,
        CancellationContext.Instance.WasCancelled ? JobState.Cancelled : JobState.Succeeded);
    Assert(JobRegistry.Instance.Query(parent.JobId)?.State == JobState.Succeeded, "terminal mapping differs");
});
await Check("C01 direct Ext group consumes gate and executes", async () =>
{
    PreemptionGate.Arm(); var ran = 0;
    var result = await new TaskRunner().RunCurrentAsync(() => { ran++; return Done(); },
        job: new(JobKind.Group, Unique(), JobSource.Ext));
    Assert(result == TaskRunResult.Ran && ran == 1 && !PreemptionGate.IsArmed, "control failed");
});
await Check("D03 unrelated V2 request consumes gate; old local workflow can then run", async () =>
{
    PreemptionGate.Arm(); var ran = 0;
    await new TaskRunner().RunCurrentAsync(Done, job: new(JobKind.Group, Unique(), JobSource.V2));
    await new TaskRunner().RunCurrentAsync(() => { ran++; return Done(); });
    Assert(!PreemptionGate.IsArmed && ran == 1, "no-owner behavior differs");
});

await Check("D07 thrown execution error still returns Ran", async () =>

await Check("C02 busy admission never executes action", async () =>
{
    await TaskControl.TaskSemaphore.WaitAsync();
    try
    {
        var ran = false;
        var result = await new TaskRunner().RunCurrentAsync(() => { ran = true; return Done(); });
        Assert(result == TaskRunResult.RejectedSlotBusy && !ran, "busy control failed");
    }
    finally { TaskControl.TaskSemaphore.Release(); }
});

    var saved = TaskContext.Instance().Config.SuspendedTaskContext!;

        job: new(JobKind.Group, saved.GroupName, JobSource.Resume));
    Assert(ran && saved.GroupName == "好感任务", "resume reproduction differs");
});

Console.WriteLine($"Characterization total={count}; passed={count - failures}; failed={failures}. PASS does not mean repaired.");
Environment.ExitCode = failures == 0 ? 0 : 1;
